\par\vspace{2pt}
\raggedright
\begin{columns}[T]
  \begin{column}{0.32\linewidth}
    \textbf{\color{marineblue}Convolution.} A filter of a few pixels is moved across the image and
    lights up wherever its pattern occurs. Each map above is one filter's response.
  \end{column}
  \begin{column}{0.32\linewidth}
    \textbf{\color{marineblue}Training.} The CNN's $\approx$\,11 million numbers are adjusted until its
    answers match the experts' labels -- learning by example, not by rules.
  \end{column}
  \begin{column}{0.32\linewidth}
    \textbf{\color{marineblue}Pretraining.} This CNN learned its filters on $\approx$\,1.3 million everyday
    photos (ImageNet); only the last step was trained on FathomNet.
  \end{column}
\end{columns}
\vfill
{\tiny\color{darkgray} Real example: ResNet-18 + logistic regression; the crop is from a dive not used in
training. 3 of the 64--512 maps per block shown (brighter = stronger response). Images: FathomNet / MBARI.}
\end{frame}

\begin{frame}{The FathomNet 2025 Challenge \,--\, and what we can do with it}
\small
\textbf{The task:} name the animal in each box -- 79 taxa at mixed ranks, $\approx$\,300 training crops each.
\textbf{The score is the distance in the tree of life}, averaged over all test animals (0 = perfect, 12 = worst).
\par\vspace{2pt}
\begin{columns}[T]
\begin{column}{0.555\linewidth}
  \textbf{\color{marineblue}Example: the truth is \emph{Mediaster aequalis}}\\[2pt]
  \exanimal{ex0}{\emph{Mediaster aequalis}}{correct}{\dist{accentgreen}{0}}\hfill
  \exanimal{ex1}{\emph{Hippasteria}}{same family}{\dist{accentgreen!60!accentorange}{3}}\hfill
  \exanimal{ex2}{Asteroidea}{the class only}{\dist{accentorange!85!accentgreen}{4}}\hfill
  \exanimal{ex3}{\emph{Rathbunaster californicus}}{other order}{\dist{accentorange}{8}}\hfill
  \exanimal{ex4}{\emph{Liponema brevicorne}}{sea anemone}{\dist{accentred}{12}}\\[4pt]
  \textbf{\color{marineblue}The winners' models are public} -- we can run and dissect them:\\[1pt]
  {\fsmall
  \setlength{\tabcolsep}{3pt}
  \begin{tabular}{@{}llr@{}}
    \toprule
    place & model -- what it looks at & score\\
    \midrule
    1st & large model (16\,GB) -- the animal \emph{and its surroundings} & 1.45\\
    3rd & 8 $\times$ WideResNet-101 -- the crop only & 1.83\\
    5th & EfficientNet + a graph of the taxonomy (60\,MB) & --\\
    \bottomrule
  \end{tabular}}
\end{column}
\begin{column}{0.41\linewidth}
  \begin{infobox}{What we already measured}
    \fsmall
    3rd and 5th place: \textbf{$\approx$\,98\,\%} correct on their training images, only
    \textbf{\textcolor{accentred}{57--61\,\%}} on dives they have never seen. Almost half of the
    3rd-place model's mistakes are in \textbf{another phylum}.
  \end{infobox}
  \vspace{1pt}
  \begin{ideabox}{Let's question the question}
    \fsmall
    \begin{itemize}\setlength{\itemsep}{1pt}
      \item Asteroidea scores 4, a wrong sea-star species 8 -- should a model rather answer cautiously?
      \item Porifera (a phylum) next to species: one task or many?
      \item Every taxon has $\approx$\,300 crops -- nature is not balanced.
      \item 1st place also looks at the habitat. What do \emph{you} use -- depth, substrate, behaviour?
    \end{itemize}
  \end{ideabox}
\end{column}
\end{columns}
\end{frame}
